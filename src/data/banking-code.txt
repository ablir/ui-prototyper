function KpiCard({ kpi }: { kpi: Kpi }) {
  return (
    <Card shadow="sm" padding="lg" radius="md" withBorder>
      <Stack gap={6}>
        <Text size="xs" c="dimmed" fw={600} tt="uppercase">
          {kpi.label}
        </Text>
        <Text fz={28} fw={700} lh={1.1}>
          {kpi.value}
        </Text>
        <Badge
          color={kpi.trend.color}
          variant="light"
          size="sm"
          radius="sm"
          leftSection={<span>{kpi.trend.direction === 'up' ? '▲' : '▼'}</span>}
        >
          {kpi.trend.pct}
        </Badge>
      </Stack>
    </Card>
  );
}

function HoldingsTable() {
  const rows = holdings.map((h) => {
    const gain = h.dayChangePct > 0;
    const flat = h.dayChangePct === 0;
    return (
      <Table.Tr key={h.ticker}>
        <Table.Td>
          <Text size="sm" fw={700}>
            {h.ticker}
          </Text>
          <Text size="xs" c="dimmed">
            {h.name}
          </Text>
        </Table.Td>
        <Table.Td>
          <Badge variant="default" size="sm" radius="sm">
            {h.assetClass}
          </Badge>
        </Table.Td>
        <Table.Td>
          <Text size="sm" fw={600}>
            <NumberFormatter prefix="$" thousandSeparator value={h.marketValue} />
          </Text>
        </Table.Td>
        <Table.Td>
          <Group gap={8} wrap="nowrap">
            <Progress value={h.weightPct * 10} color="indigo" size="sm" radius="xl" w={70} />
            <Text size="sm" w={42} ta="right">
              {h.weightPct.toFixed(1)}%
            </Text>
          </Group>
        </Table.Td>
        <Table.Td>
          <Badge
            color={flat ? 'gray' : gain ? 'teal' : 'red'}
            variant="light"
            size="sm"
            radius="sm"
            leftSection={<span>{flat ? '—' : gain ? '▲' : '▼'}</span>}
          >
            {h.dayChangePct > 0 ? '+' : ''}
            {h.dayChangePct.toFixed(2)}%
          </Badge>
        </Table.Td>
      </Table.Tr>
    );
  });

  return (
    <ScrollArea>
      <Table highlightOnHover withTableBorder verticalSpacing="sm" miw={620}>
        <Table.Thead>
          <Table.Tr>
            <Table.Th>Position</Table.Th>
            <Table.Th>Asset Class</Table.Th>
            <Table.Th>Market Value</Table.Th>
            <Table.Th>Weight</Table.Th>
            <Table.Th>Day Change</Table.Th>
          </Table.Tr>
        </Table.Thead>
        <Table.Tbody>{rows}</Table.Tbody>
      </Table>
    </ScrollArea>
  );
}

function TransactionsTimeline() {
  return (
    <Paper shadow="sm" radius="md" withBorder p="lg" h="100%">
      <Title order={4} mb="lg">
        Recent Transactions
      </Title>
      <Timeline active={transactions.length} bulletSize={28} lineWidth={2}>
        {transactions.map((tx, idx) => {
          const cfg = txConfig[tx.type];
          return (
            <Timeline.Item
              key={idx}
              bullet={
                <ThemeIcon color={cfg.color} size={28} radius="xl" variant="light">
                  <span style={{ fontSize: 14, fontWeight: 700 }}>{cfg.glyph}</span>
                </ThemeIcon>
              }
              title={
                <Group gap="xs">
                  <Text size="sm" fw={600}>
                    {cfg.label}
                  </Text>
                  <Badge color={cfg.color} variant="light" size="xs" radius="sm">
                    {tx.date}
                  </Badge>
                </Group>
              }
            >
              <Text size="sm" c="dimmed">
                {tx.description}
              </Text>
            </Timeline.Item>
          );
        })}
      </Timeline>
    </Paper>
  );
}
